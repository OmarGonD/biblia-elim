# building Ministerios Elim

message (STATUS "Configuring Ministerios Elim target")

# compile the embedded GtkBuilder UI files into
# a C source linked directly into the Ministerios Elim binary. See issue #894
# item 4 - avoids looking up .gtkbuilder files on disk at runtime.
find_program (GLIB_COMPILE_RESOURCES NAMES glib-compile-resources REQUIRED)

set (biblia_elim_gresource_xml ${PROJECT_SOURCE_DIR}/ui/biblia-elim.gresource.xml)
set (biblia_elim_resources_c ${CMAKE_CURRENT_BINARY_DIR}/biblia_elim_resources.c)
set (biblia_elim_resources_dep ${CMAKE_CURRENT_BINARY_DIR}/biblia_elim_resources.c.d)

# ask glib-compile-resources for the list of files the manifest
# references, so the custom command re-runs whenever any of the
# embedded .gtkbuilder/.ui files change, not just the manifest itself.
execute_process (
  COMMAND ${GLIB_COMPILE_RESOURCES}
          --sourcedir=${PROJECT_SOURCE_DIR}/ui
          --generate-dependencies
          ${biblia_elim_gresource_xml}
  OUTPUT_VARIABLE biblia_elim_gresource_deps_raw
  RESULT_VARIABLE biblia_elim_gresource_deps_result
  )
if (NOT biblia_elim_gresource_deps_result EQUAL 0)
  message (FATAL_ERROR "Failed to list dependencies of ${biblia_elim_gresource_xml}")
endif ()
string (STRIP "${biblia_elim_gresource_deps_raw}" biblia_elim_gresource_deps_raw)
string (REPLACE "\n" ";" biblia_elim_gresource_deps "${biblia_elim_gresource_deps_raw}")

add_custom_command (
  OUTPUT ${biblia_elim_resources_c}
  COMMAND ${GLIB_COMPILE_RESOURCES}
          --sourcedir=${PROJECT_SOURCE_DIR}/ui
          --target=${biblia_elim_resources_c}
          --generate-source
          ${biblia_elim_gresource_xml}
  DEPENDS ${biblia_elim_gresource_xml} ${biblia_elim_gresource_deps}
  COMMENT "Generating Ministerios Elim resources"
  VERBATIM
  )
set_source_files_properties(${biblia_elim_resources_c} PROPERTIES GENERATED 1)

# generate locale_set.c:

# convert po/LINGUAS file into a C source file containing full locale names.
# names in LINGUAS are either 5-char full "language_LOCALE" names or
# 2-char "language-only" names.  convert the latter.

# This section is, for portability reasons, a rewrite in CMake code
# of src/gtk/make-locale-set.sh.

# 1. read po/LINGUAS
file (READ ${PROJECT_SOURCE_DIR}/po/LINGUAS linguas)
# transform list of languages in a regular CMake list
string (REGEX MATCHALL "[a-zA-Z_]+" langs "${linguas}")

# 2. create header of locate_set.c
set (locale_set ${CMAKE_CURRENT_BINARY_DIR}/locale_set.c)
file (WRITE ${locale_set} "
/* File generated by CMake - Do not edit! */
char *locale_set[] = {\n"
  )
# 3. insert locale names
foreach(LANG ${langs})
  # special case, english.
  if (LANG STREQUAL "en_GB")
    file (APPEND ${locale_set} "  \"en_GB\",\n")
    file (APPEND ${locale_set} "  \"en_US\",\n")
  elseif (LANG STREQUAL "cs")
    # special case, czech.
    file (APPEND ${locale_set} "  \"cs_CZ\",\n")
  elseif (LANG STREQUAL "fa")
    # special case, iran.
    file (APPEND ${locale_set} "  \"fa_IR\",\n")
  elseif(LANG STREQUAL "he")
    # special case, israel.
    file (APPEND ${locale_set} "  \"he_IL\",\n")
  elseif(LANG STREQUAL "nb")
    # special case, norway.
    file (APPEND ${locale_set} "  \"nb_NO\",\n")
  elseif(LANG STREQUAL "sl")
    # special case, slovenia.
    file (APPEND ${locale_set} "  \"sl_SI\",\n")
  elseif(LANG STREQUAL "sv")
    # special case, sweden.
    file (APPEND ${locale_set} "  \"sv_SE\",\n")
  else ()
    string (LENGTH ${LANG} LANG_LEN)
    if (LANG_LEN LESS 5)
      # special case, 2-char "language-only" names.
      string (TOUPPER ${LANG} UPPERLANG)
      file (APPEND ${locale_set} "  \"${LANG}_${UPPERLANG}\",\n")
    else ()
      # common 5-char full "language_LOCALE" names
      file (APPEND ${locale_set} "  \"${LANG}\",\n")
    endif ()
  endif ()
endforeach ()
# 4. add footer to locale_set.c
file (APPEND ${locale_set} "  (char*)0
};\n"
  )
# 5. locale_set.c is tagged as generated during the build
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/locale_set.c PROPERTIES GENERATED 1)

if (DBUS)
  # need to generate 4 files using gdbus-codegen:
  # marshal.c, marshal.h, ipc-gdbus.c, ipc-gdbus.h
  add_custom_command (
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/marshal.h
    ${CMAKE_CURRENT_BINARY_DIR}/marshal.c
    ${CMAKE_CURRENT_BINARY_DIR}/ipc-gdbus.h
    ${CMAKE_CURRENT_BINARY_DIR}/ipc-gdbus.c

    # process marshal.list to generate marshal.h
    COMMAND ${GLIB_GENMARSHAL}
    ${CMAKE_CURRENT_SOURCE_DIR}/marshal.list
    --prefix=ipc_marshal
    --header > ${CMAKE_CURRENT_BINARY_DIR}/marshal.h
    MAIN_DEPENDENCY ${CMAKE_CURRENT_SOURCE_DIR}/marshal.list

    # process marshal.list to generate marshal.c
    COMMAND ${GLIB_GENMARSHAL}
    ${CMAKE_CURRENT_SOURCE_DIR}/marshal.list
    --prefix=ipc_marshal
    --body > ${CMAKE_CURRENT_BINARY_DIR}/marshal.c
    MAIN_DEPENDENCY ${CMAKE_CURRENT_SOURCE_DIR}/marshal.list

    # process ipc-interface.xml with gdbus-codegen to generate GDBus bindings
    COMMAND gdbus-codegen
    --interface-prefix org.bibliaelim.
    --generate-c-code ipc-gdbus
    --c-namespace BibliaElim
    ${CMAKE_CURRENT_SOURCE_DIR}/ipc-interface.xml
    MAIN_DEPENDENCY ${CMAKE_CURRENT_SOURCE_DIR}/ipc-interface.xml

    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
    COMMENT "Generating src/gtk D-Bus bindings"
    )

  # add files to the sources
  target_sources (biblia-elim
    PRIVATE
    ipc.c
    ${CMAKE_CURRENT_BINARY_DIR}/marshal.c
    ${CMAKE_CURRENT_BINARY_DIR}/ipc-gdbus.c
    )
  target_include_directories (biblia-elim
    PRIVATE ${CMAKE_CURRENT_BINARY_DIR}
    )
endif (DBUS)
